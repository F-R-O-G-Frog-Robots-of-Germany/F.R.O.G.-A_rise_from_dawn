  \underbrace{\vphantom{\displaystyle\sum\frac{\Delta e}{\Delta t}}#1}%
  _{\text{\color{secondary}#2}}%
}

  \begin{minipage}{94mm}

  Error \(e = \text{target speed} - \text{measured speed}\)
  {\color{secondary}(encoder ticks/s)}.\par
  \[
    \text{PIDF} =
    \term{K_P e}{Proportional}
    + \term{K_I \sum e\,\Delta t}{Integral}
    + \term{K_D \frac{\Delta e}{\Delta t}}{Derivative}
    + \term{K_F \times \text{target speed}}{Feedforward}
  \]

  \[
    \text{\bfseries Motor output} =
    \begin{cases}
      0, & \text{zero target or above target}, \\
      1, & \text{more than }150\,\text{RPM below target}, \\
      \text{PIDF}, & \text{otherwise}.
    \end{cases}
  \]
  \end{minipage}%
